cmake_minimum_required(VERSION 3.16)
project(capp VERSION 0.1.0 LANGUAGES C)
include(CTest)
include(GNUInstallDirs)

add_library(app_core STATIC src/app.c)
target_include_directories(app_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(app_core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(capp src/main.c)
target_link_libraries(capp PRIVATE app_core)
set_target_properties(capp PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)

function(app_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
app_warnings(app_core)
app_warnings(capp)
if(BUILD_TESTING)
    add_executable(test_app tests/test_app.c)
    target_link_libraries(test_app PRIVATE app_core)
    set_target_properties(test_app PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    app_warnings(test_app)
    add_test(NAME app_codec COMMAND test_app)
endif()
install(TARGETS capp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
